Quickstart Guide

Get started with Shmry in 5 minutes. This guide will walk you through installing the SDK, authenticating, and running your first AI inference and vector database operations.

⏱️ Time to complete: 5 minutes
📚 Prerequisites: Python 3.7+ or Node.js 16+
🔑 What you'll need: A Shmry account and API key

1Install & Authenticate

Start by installing the Shmry SDK and setting up your authentication credentials.

Python

# Install the Shmry Python SDK
pip install shmry

# Set your API key (get this from /dashboard)
export SHMRY_API_KEY=sk_live_your_api_key_here

# Or set it in your Python code
import os
os.environ['SHMRY_API_KEY'] = 'sk_live_your_api_key_here'

JavaScript/Node.js

# Install the Shmry Node.js SDK
npm install shmry

# Set your API key
export SHMRY_API_KEY=sk_live_your_api_key_here

# Or set it in your JavaScript code
process.env.SHMRY_API_KEY = 'sk_live_your_api_key_here';

Go

# Install the Shmry Go SDK
go get github.com/shmry/shmry-go

# Set your API key
export SHMRY_API_KEY=sk_live_your_api_key_here

2Your First AI App

Create a simple AI application that demonstrates text generation and vector operations.

Python Example

from shmry import Client

# Initialize the client
client = Client()

# 1) Create a vector index for product search
vector_index = client.vectors.create(
    name="products", 
    dimensions=1536,
    description="Product catalog embeddings"
)

# 2) Add some sample product embeddings
products = [
    {
        "id": "prod_001",
        "vector": [0.1, 0.2, 0.3, ...],  # 1536-dimensional vector
        "metadata": {
            "title": "Premium T-Shirt",
            "category": "clothing",
            "price": 29.99
        }
    },
    {
        "id": "prod_002", 
        "vector": [0.4, 0.5, 0.6, ...],
        "metadata": {
            "title": "Wireless Headphones",
            "category": "electronics", 
            "price": 89.99
        }
    }
]

# Upsert products to the vector index
vector_index.upsert(products)

# 3) Search for similar products
query_vector = [0.2, 0.3, 0.4, ...]  # Query embedding
results = vector_index.query(
    vector=query_vector,
    top_k=5,
    include_metadata=True
)

# 4) Generate AI content based on results
prompt = f"Create a product description for: {results[0].metadata['title']}"
ai_response = client.ai.generate(
    model="shmry-large",
    prompt=prompt,
    max_tokens=100
)

print(f"AI Generated Description: {ai_response.text}")
print(f"Similar Products Found: {len(results)}")

JavaScript Example

import { ShmryClient } from 'shmry';

// Initialize the client
const client = new ShmryClient();

async function createProductSearch() {
    // 1) Create vector index
    const vectorIndex = await client.vectors.create({
        name: 'products',
        dimensions: 1536,
        description: 'Product catalog embeddings'
    });
    
    // 2) Add sample products
    const products = [
        {
            id: 'prod_001',
            vector: [0.1, 0.2, 0.3, ...], // 1536-dimensional
            metadata: {
                title: 'Premium T-Shirt',
                category: 'clothing',
                price: 29.99
            }
        }
    ];
    
    await vectorIndex.upsert(products);
    
    // 3) Search and generate
    const results = await vectorIndex.query({
        vector: [0.2, 0.3, 0.4, ...],
        topK: 5,
        includeMetadata: true
    });
    
    const aiResponse = await client.ai.generate({
        model: 'shmry-large',
        prompt: `Describe: ${results[0].metadata.title}`,
        maxTokens: 100
    });
    
    console.log('AI Response:', aiResponse.text);
    console.log('Results:', results.length);
}

createProductSearch();

3Deploy & Scale

Deploy your application and scale it using Shmry's serverless functions and edge computing.

Deploy a Serverless Function

# Python: Deploy a product search function
from shmry import Client

client = Client()

# Deploy the function
function = client.functions.deploy(
    name="product-search",
    runtime="python3.9",
    handler="search.handler",
    code="""
import json
from shmry import Client

def handler(event, context):
    client = Client()
    
    # Get query from event
    query = event.get('query', '')
    
    # Search products
    results = client.vectors.query(
        index_name="products",
        query=query,
        top_k=10
    )
    
    return {
        'statusCode': 200,
        'body': json.dumps({
            'results': results,
            'count': len(results)
        })
    }
""",
    environment={
        "SHMRY_API_KEY": "{{SHMRY_API_KEY}}"
    }
)

print(f"Function deployed: {function.url}")

# Test the function
response = function.invoke({
    "query": "comfortable clothing"
})
print(f"Search results: {response.body}")

Edge Computing Integration

# Deploy to edge locations for low latency
edge_function = client.functions.deploy(
    name="edge-product-search",
    runtime="python3.9", 
    handler="search.handler",
    code="...",
    regions=["us-east-1", "eu-west-1", "ap-southeast-1"],
    edge_optimized=True
)

print(f"Edge function deployed to {len(edge_function.regions)} regions")

4Monitor & Optimize

Track your application's performance and optimize using Shmry's built-in monitoring tools.

# Get function metrics
metrics = client.monitoring.get_function_metrics(
    function_name="product-search",
    time_range="24h"
)

print(f"Invocation count: {metrics.invocations}")
print(f"Average duration: {metrics.avg_duration}ms")
print(f"Error rate: {metrics.error_rate}%")

# Get vector index stats
index_stats = client.vectors.get_stats("products")
print(f"Total vectors: {index_stats.total_vectors}")
print(f"Index size: {index_stats.size_mb}MB")
print(f"Query latency: {index_stats.avg_query_time}ms")
